    size_category = case_when(
      baseline_overall >= 1000 & baseline_overall < 5000 ~ "1k-5k",

      baseline_overall >= 10000 & baseline_overall < 20000 ~ "10k-20k",

      labels = c("R1", "R2", "D/PU", "M", "B", "SF&TCU")
    )
  ) %>%
  left_join(
    y = tibble(
      unit_id = rownames(coef(m_public_1)$unit_id) %>% as.numeric(),
      time_slope = coef(m_public_1)$unit_id[,"x_year"],
      enrl_intcpt = coef(m_public_1)$unit_id[,"(Intercept)"],
      growth_pct = time_slope / enrl_intcpt,
      growth_bucket = case_when(
        growth_pct < -0.05 ~ "Linear Decline more than 5% per year",
        growth_pct < -0.01 ~ "Linear Decline 1 to 5% per year",
        growth_pct <= 0.01 ~ "Linear Change less than 1% per year",

    ) ,
    by = "unit_id"
  )
