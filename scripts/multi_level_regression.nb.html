public_dat3_x <- public_dat3 %>%
  # In the Carnegie universe
  filter(carnegie != "Not Carnegie") %>%
  # At least 1000 overall enrollment in first reporting year
  group_by(unit_id) %>%
  filter(year == min(year)) %>%
  ungroup() %>%
  filter(overall_enrollment >= 1000) %>%
  distinct(unit_id) %>%
  left_join(
    y = public_dat3,
    by = "unit_id"
  ) %>%
  # Has all 5 years of data
  group_by(unit_id) %>%
  summarise(N_year = length(unique(year))) %>%
  ungroup() %>%
  left_join(
    y = public_dat3,
    by = "unit_id"
  ) %>%
  mutate(
    year_0 = year - min(year),
    # Log transforms: first-time enrollment; HS grads by state; applicants; average net price with aid

Model 1: Basic Growth Model (Null Model)

formula_1 <- formula(
  first_time_enrollment ~
    1
    + x_year
    + (1 + x_year | unit_id)
)
