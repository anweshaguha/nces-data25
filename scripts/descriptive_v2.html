admissions_data <- import(here("Data", "Data Admissions Variables.csv"))

# 1: Select only total count columns (exclude imputation flags)
admission_data_clean <- admissions_data %>%
  clean_names() %>% 
  select(unit_id, institution_name, matches("^applicants|^admissions|^enrolled")) %>%
  pivot_longer(
    cols = -c(unit_id, institution_name),
    names_to = "variable",
    values_to = "value"
  ) %>%
  
  # 2: Extract category and year from column names
  mutate(
    category = case_when(
      str_detect(variable, "applicants") ~ "applicants",
      str_detect(variable, "admissions") ~ "admissions",
      str_detect(variable, "enrolled") ~ "enrolled"
    ),
    year = str_extract(variable, "20\\d{2}"),
    year = as.integer(year)
  ) %>%
  filter(!is.na(year)) %>%  # Remove NA years (some might not match pattern)
  filter(value > 1) # Should have over 1 student enrolled

summary_stats <- admission_data_clean %>%
  group_by(year, category) %>%
  summarise(
    Mean = round(mean(value, na.rm = TRUE)),
    Median = round(median(value, na.rm = TRUE)),
    SD = round(sd(value, na.rm = TRUE), 2),
    Count = n()
  )

3.4 Percent of undergraduate students awarded Federal Pell grants (numeric: %, 0 to 100)

studentbody_data <- import(here("Data", "Data Student Body Composition Variables.csv"))

studentbody_data <- studentbody_data %>% 
  select(-V88, -contains("Imputation")) %>% 
  pivot_longer(
    cols = -c(1:2),
    names_to = "VariableName",
    values_to = "Value"
  )

studentbody_data_clean <- studentbody_data %>% 
  rename_with(~ str_trim(.), everything()) %>%  # trim any weird spacing in names
  mutate(
    year = case_when(
      str_detect(VariableName, "DRVEF\\d{4}") ~ str_extract(VariableName, "DRVEF(\\d{4})") %>% str_remove("DRVEF"),
      str_detect(VariableName, "SFA\\d{4}") ~ paste0("20", str_sub(str_extract(VariableName, "SFA\\d{4}"), -2)),  # grab last 2 digits
      TRUE ~ NA_character_
    ),  # extract 4-digit year conditionally
    clean_label = str_remove(VariableName, "\\s*\\([^\\)]+\\)")  # remove everything in parentheses
  ) %>% #looks good! I'll remove original variable name
  select(-VariableName)

# The Pell and Non-resident variables are broken into two different variables due to different naming conventions. I'll combine those.

studentbody_data_clean <- studentbody_data_clean %>% 
  mutate(clean_label = recode(clean_label, "Percent of undergraduate enrollment that are Nonresident Alien" = 
        "Percent of undergraduate enrollment that are U.S. Nonresident",
        "Percent of undergraduate students awarded Pell grants" = 
        "Percent of undergraduate students awarded Federal Pell grants"))

# table(studentbody_data_clean$clean_label) #check

minoritized_dat <- studentbody_data_clean %>% 
  mutate(clean_label = case_when(clean_label == "Percent of undergraduate students awarded Federal Pell grants" ~ "pct_pell",
                               clean_label == "Percent of undergraduate enrollment that are White" ~ "pct_white",
                               clean_label == "Undergraduate enrollment" ~ "enrollment")) %>% 
  filter(clean_label == "pct_pell" | clean_label == "pct_white" | clean_label == "enrollment") %>% 
  janitor::clean_names() %>% 
  pivot_wider(names_from = clean_label,
              values_from = value) %>% 
  mutate(pct_race_minority = 100 - pct_white) %>% 
  select(-pct_white) %>% 
  filter(enrollment > 1)

wiche_data_allrows <- import(here("Data/Non-IPEDS Data/WICHE HS Graduates by State.xlsx"), which = "Data")


hs_grads_by_year <- wiche_data_allrows %>%
  filter(
    Grade == "High school graduates", # Keep only rows ~ total high school graduates
    RaceEthnicity == "Total/any", # Keep only rows ~ total all race/ethnicity
    SchoolSector == "Grand Total (public+private)", # Keep only rows ~ total public and private high schools
    !(str_detect(string = Stabbr, pattern = "^_")), # Keep only rows NOT starting with an underscore (i.e., drop region aggregates)
    !(str_detect(string = Stabbr, pattern = "\\d")), # Keep only rows NOT containing digits (i.e., drop region aggregates)
    !(Stabbr %in% c("CM","CW")) # Keep only rows NOT CM and CW (i.e., drop region aggregates)
  ) %>%
  mutate(
    HS_Grad_Year = as.numeric(str_extract(string = ClassOf, pattern = "\\d+")) # Extract the graduation year for numeric filtering
  ) %>%
  filter(
    HS_Grad_Year >= 2017,
    HS_Grad_Year < 2026
  ) %>%
  select(
    # Keep only these variables
    Stabbr,
    HS_Grad_Year,
    Students
  ) %>%
  # Convert to wide format...
  pivot_wider(id_cols = Stabbr, names_from = HS_Grad_Year, values_from = Students, names_prefix = "HS_Grad_") %>% 
  janitor::clean_names() %>% 
  select(stabbr, hs_grad_2018:hs_grad_2022)
